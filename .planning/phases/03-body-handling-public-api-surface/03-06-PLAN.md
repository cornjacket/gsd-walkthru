    ```ts
    //

    // RequestHandler compatible with both Express 4.x and 5.x.
    //
    // Call-time invariants (fail loudly per PITFALLS #11):
    //   - D-02: providerName must be in the registry → plain Error if not.

    //   - D-07: req.rawBody MUST be populated upstream by rawBodyCapture()

    // logic. The factory only knows how to look up a provider by string and
    // call its Provider interface (D-01). Phase 4/5 replace each stub
    // provider's validate() method without touching this factory.
    //
    // D-16: zero console.* calls. The error class (Phase 2 D-11) plus
    // webhookErrorHandler() (Plan 04) are the only failure-surface
    // components; logging is the consumer's choice.
    import type { RequestHandler } from 'express';
    import { WebhookValidationError } from './errors.js';
    import { getProvider } from './providers/registry.js';

    export interface CreateWebhookMiddlewareOptions {
      /** The webhook signing secret. Must be non-empty (PITFALLS #11). */
      secret: string;
    }

    /**
     * The single primary library export (API-01). Looks up `providerName`
     * in the registry, validates configuration synchronously, and returns
     * an Express RequestHandler that:
     *   - Yields `WebhookValidationError({ reason: 'malformed_payload' })`
     *     when req.rawBody is missing.
     *   - Calls provider.validate(req, options.secret), assigns the result
     *     to req.webhook, and calls next() on success.
     *   - Delegates any thrown error from validate() via next(err).
     *
     * @throws Error - synchronously at call time when providerName is not
     *                  registered (D-02) or options.secret is empty (PITFALLS #11).
     */
    export function createWebhookMiddleware(
      providerName: string,
      options: CreateWebhookMiddlewareOptions,
    ): RequestHandler {
      // D-02: synchronous validation at call time, not request time.
      const provider = getProvider(providerName);
      if (!provider) {
        // Static list of v1 provider names for a helpful error message.
        // Phase 4/5 REPLACE these entries (they don't add new names), so a
        // dynamic lookup of registered names from registry.ts is unnecessary
        // for v1.
        const known = ['stripe', 'github', 'shopify'].join(', ');
        throw new Error(
          `Unknown webhook provider: '${providerName}'. ` +
            `Registered v1 providers: ${known}.`,
        );
      }

      // PITFALLS #11 / T-3-03: fail loudly on missing/empty secret.
      // D-04: plain Error for config issues, NOT WebhookValidationError.
      if (!options.secret) {
        throw new Error(
          `Webhook secret required for provider '${providerName}'`,
        );
      }

      const secret = options.secret;

      return (req, _res, next) => {
        // D-07: missing rawBody → malformed_payload validation error.
        if (!req.rawBody) {
          return next(
            new WebhookValidationError({
              reason: 'malformed_payload',
              provider: providerName,
              statusCode: 400,
            }),
          );
        }

        try {
          const metadata = provider.validate(req, secret);
          req.webhook = metadata;
          next();
        } catch (err) {
          // D-13: never write res ourselves; delegate via next(err).
          next(err);
        }
      };
    }
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Webhook provider (Stripe/GitHub/Shopify) → Express request → middleware | Untrusted payload + headers cross from network into the validation factory's returned handler. |
| Factory call site (consumer's app startup) → registry lookup | Consumer-supplied `providerName` and `secret` cross from app config into the factory at call time. |
| Provider.validate() throws → next(err) → consumer error pipeline | Validation failures cross from the middleware into webhookErrorHandler() (Plan 04) or the consumer's own error pipeline. |
| Missing rawBody → WebhookValidationError({ reason: 'malformed_payload' }) | An upstream misconfiguration (no rawBodyCapture mounted) yields a structured 400 error rather than an unhandled crash. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-3-03 | Tampering / Elevation | Fail-open on missing/empty secret at factory call time | mitigate | `if (!options.secret)` synchronous throw in `createWebhookMiddleware` (PITFALLS #11). Plain `Error` per D-04 — surfaces in CI/deploy, never under live traffic. Test `throws synchronously on empty secret` locks this contract. The factory throws BEFORE returning the handler, so no request can reach the handler with an empty secret. |
| T-3-01 | Information Disclosure | Signature/secret/body leakage in error responses | mitigate | Middleware NEVER writes to `res` directly (D-13) — only `next(err)`. The error class has no field for sensitive data (Phase 2 D-11 structural guarantee, extended in Plan 04 to the new `'malformed_payload'` reason). Test `response body never contains sample signature/secret/body bytes` is the API-04 / SC4 acceptance test for the full request→middleware→error-handler→response chain. |
| T-3-05 | Information Disclosure | Library logging on validation paths | mitigate | Source file has zero `console.*` references (acceptance grep). Test `calls no logging methods on success or failure paths` uses `vi.spyOn(console, ...)` and asserts 0 calls across both end-to-end paths. |

Test `delegates provider.validate() throws via next(err)` confirms `req.webhook` is not assigned (the downstream handler doesn't run). |
